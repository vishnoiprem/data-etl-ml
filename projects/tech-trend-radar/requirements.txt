# Tech Trend Radar — dependencies
#
# House style: lower bounds only, with a comment on why each one is here.
# Everything runs on the CPU with no API key and no Docker.

# HTTP client for the source adapters. One client, reused across sources,
# so connection pooling and timeouts are configured in exactly one place.
httpx>=0.27

# The API layer. FastAPI gives the structured response the project is
# built around — a request in, a ranked trend list out.
fastapi>=0.110

# Request validation and the response models the API returns.
pydantic>=2.6

# ASGI server for `make api`.
uvicorn>=0.27

# Test runner and linter. CI installs only this file before running
# `python -m pytest` and `ruff check`, so both have to be listed here
# (the same arrangement as copyforge-ai and trade-sourcing-ai).
pytest>=8.3
ruff>=0.6

# RSS parsing is done with the standard library (xml.etree), so no feed
# parser dependency is needed. Adding one would be the only reason to
# touch the network for data the stdlib already handles.

# The semantic topic matcher lives in a separate file because it is the
# one heavy dependency in the project — `sentence-transformers` pulls in
# torch, which is a ~2 GB install and would dominate CI.
#
# It is *optional*: the radar degrades to lexical-only extraction when it
# is absent, and the tests that need it skip themselves. Install it with
# `make up-semantic` for the full pipeline.
#
#   sentence-transformers>=3.0
