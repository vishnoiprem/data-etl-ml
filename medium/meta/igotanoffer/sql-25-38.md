## Problem 25 — Top 10 colleges/companies an "average social person" interacts with

**Interpretation:** Rank interactions by frequency, return top 10 entities. "Average social person" = the median or mean interaction count per user — used as a baseline.

### Approach 1 — GROUP BY + ORDER BY + LIMIT
```sql
SELECT entity, COUNT(*) AS interactions
FROM interactions
GROUP BY entity
ORDER BY interactions DESC
LIMIT 10;
```

### Approach 2 — CTE with rank + filter
```sql
WITH counts AS (
  SELECT entity, COUNT(*) AS interactions
  FROM interactions
  GROUP BY entity
),
ranked AS (
  SELECT entity, interactions, ROW_NUMBER() OVER (ORDER BY interactions DESC) AS rk
  FROM counts
)
SELECT entity, interactions FROM ranked WHERE rk <= 10;
```

### Approach 3 — DENSE_RANK with tie handling
```sql
SELECT entity, interactions FROM (
  SELECT entity, COUNT(*) AS interactions,
         DENSE_RANK() OVER (ORDER BY COUNT(*) DESC) AS rk
  FROM interactions
  GROUP BY entity
) t WHERE rk <= 10;
```

### Approach 4 — subquery with IN
```sql
SELECT * FROM interactions
WHERE entity IN (
  SELECT entity FROM interactions
  GROUP BY entity
  ORDER BY COUNT(*) DESC LIMIT 10
);
```

### Approach 5 — HAVING on aggregate
```sql
SELECT entity, COUNT(*) AS interactions
FROM interactions
GROUP BY entity
HAVING COUNT(*) >= (
  SELECT MIN(c) FROM (
    SELECT COUNT(*) AS c FROM interactions GROUP BY entity
    ORDER BY c DESC LIMIT 10
  ) x
);
```

### Approach 6 — window with NTILE to bucket
```sql
SELECT entity, interactions FROM (
  SELECT entity, COUNT(*) AS interactions,
         NTILE(100) OVER (ORDER BY COUNT(*) DESC) AS bucket
  FROM interactions GROUP BY entity
) t WHERE bucket = 1;
```

### Approach 7 — JOIN with aggregates
```sql
SELECT i.entity, COUNT(*) AS interactions
FROM interactions i
JOIN users u ON u.user_id = i.user_id
WHERE u.activity >= (SELECT AVG(activity) FROM users)
GROUP BY i.entity
ORDER BY interactions DESC LIMIT 10;
```

### Approach 8 — LATERAL + LIMIT
```sql
SELECT entity, interactions FROM (
  SELECT entity, COUNT(*) AS interactions
  FROM interactions GROUP BY entity
) c
ORDER BY interactions DESC LIMIT 10;
```

### Approach 9 — DISTINCT ON (PostgreSQL)
```sql
SELECT DISTINCT ON (entity) entity, COUNT(*) OVER (PARTITION BY entity) AS interactions
FROM interactions
ORDER BY entity, interactions DESC LIMIT 10;
```

### Approach 10 — pandas equivalent
```python
import pandas as pd
def top_entities(df, n=10):
    return (df.groupby('entity').size()
              .sort_values(ascending=False).head(n).reset_index(name='interactions'))
```

**Best:** Approach 2 — clean CTE, easy to add tie-breaking / partition filters.

## Problem 26 — Average friend requests sent in the last week

`requests(user_id, friend_id, requested_at)`

### Approach 1 — aggregate over filter
```sql
SELECT AVG(c) AS avg_requests
FROM (
  SELECT user_id, COUNT(*) AS c
  FROM requests
  WHERE requested_at >= CURRENT_DATE - INTERVAL '7 days'
  GROUP BY user_id
) t;
```

### Approach 2 — total / distinct users
```sql
SELECT COUNT(*)::FLOAT / COUNT(DISTINCT user_id) AS avg_requests
FROM requests
WHERE requested_at >= CURRENT_DATE - INTERVAL '7 days';
```

### Approach 3 — CTE
```sql
WITH weekly AS (
  SELECT user_id, COUNT(*) AS c
  FROM requests
  WHERE requested_at >= CURRENT_DATE - INTERVAL '7 days'
  GROUP BY user_id
)
SELECT AVG(c) AS avg_requests FROM weekly;
```

### Approach 4 — window function per user
```sql
SELECT DISTINCT user_id,
       COUNT(*) OVER (PARTITION BY user_id) AS requests,
       AVG(COUNT(*) OVER (PARTITION BY user_id)) OVER () AS avg_requests
FROM requests
WHERE requested_at >= CURRENT_DATE - INTERVAL '7 days';
```

### Approach 5 — subquery in SELECT
```sql
SELECT (SELECT COUNT(*) FROM requests
        WHERE requested_at >= CURRENT_DATE - INTERVAL '7 days')::FLOAT
     / (SELECT COUNT(DISTINCT user_id) FROM requests
        WHERE requested_at >= CURRENT_DATE - INTERVAL '7 days') AS avg_requests;
```

### Approach 6 — HAVING only active users
```sql
WITH weekly AS (
  SELECT user_id, COUNT(*) c
  FROM requests
  WHERE requested_at >= CURRENT_DATE - INTERVAL '7 days'
  GROUP BY user_id
  HAVING COUNT(*) > 0
)
SELECT AVG(c) FROM weekly;
```

### Approach 7 — date_trunc
```sql
SELECT AVG(c) FROM (
  SELECT user_id, COUNT(*) c
  FROM requests
  WHERE requested_at >= date_trunc('week', CURRENT_DATE) - INTERVAL '7 days'
  GROUP BY user_id
) t;
```

### Approach 8 — INTERVAL cast
```sql
SELECT AVG(c) FROM (
  SELECT user_id, COUNT(*) c
  FROM requests
  WHERE requested_at >= NOW() - INTERVAL '7 days'
  GROUP BY user_id
) t;
```

### Approach 9 — EXISTS to filter active only
```sql
SELECT AVG(c) FROM (
  SELECT user_id, COUNT(*) c
  FROM requests r
  WHERE r.requested_at >= CURRENT_DATE - 7
    AND EXISTS (SELECT 1 FROM requests r2 WHERE r2.user_id = r.user_id)
  GROUP BY user_id
) t;
```

### Approach 10 — pandas equivalent
```python
import pandas as pd
def avg_requests(df):
    cutoff = pd.Timestamp.now() - pd.Timedelta(days=7)
    weekly = df[df['requested_at'] >= cutoff]
    return weekly.groupby('user_id').size().mean()
```

**Best:** Approach 1 — single CTE with clear semantics.

## Problem 27 — Books in good condition and not returned; % renewed >2 times

`books(book_id, title, condition)`, `loans(book_id, user_id, due_date, return_date, renewal_count)`

### Approach 1 — JOIN + CASE for both metrics
```sql
SELECT
  COUNT(*) FILTER (WHERE b.condition = 'good' AND l.return_date IS NULL) AS good_unreturned,
  100.0 * COUNT(*) FILTER (WHERE l.renewal_count > 2 AND b.condition = 'good' AND l.return_date IS NULL)
         / NULLIF(COUNT(*) FILTER (WHERE b.condition = 'good' AND l.return_date IS NULL), 0) AS pct_renewed
FROM books b
JOIN loans l ON l.book_id = b.book_id;
```

### Approach 2 — CTE + subquery
```sql
WITH good_unreturned AS (
  SELECT l.loan_id, l.renewal_count
  FROM loans l
  JOIN books b ON b.book_id = l.book_id
  WHERE b.condition = 'good' AND l.return_date IS NULL
)
SELECT
  COUNT(*) AS total_good_unreturned,
  100.0 * COUNT(*) FILTER (WHERE renewal_count > 2) / COUNT(*) AS pct_renewed_gt_2
FROM good_unreturned;
```

### Approach 3 — UNION of two aggregations
```sql
SELECT 'good_unreturned' AS metric, COUNT(*) AS value
FROM books b JOIN loans l ON l.book_id = b.book_id
WHERE b.condition = 'good' AND l.return_date IS NULL
UNION ALL
SELECT 'pct_renewed', 100.0 * COUNT(*) FILTER (WHERE l.renewal_count > 2) / COUNT(*)
FROM books b JOIN loans l ON l.book_id = b.book_id
WHERE b.condition = 'good' AND l.return_date IS NULL;
```

### Approach 4 — window function + filter
```sql
WITH base AS (
  SELECT l.renewal_count,
         COUNT(*) OVER () AS total,
         COUNT(*) OVER (PARTITION BY l.renewal_count > 2) AS bucket
  FROM loans l JOIN books b ON b.book_id = l.book_id
  WHERE b.condition = 'good' AND l.return_date IS NULL
)
SELECT MAX(total) AS total_good_unreturned,
       100.0 * MAX(bucket) / MAX(total) AS pct_renewed
FROM base;
```

### Approach 5 — HAVING
```sql
SELECT COUNT(*) AS total, 100.0 * SUM(CASE WHEN renewal_count > 2 THEN 1 ELSE 0 END) / COUNT(*) AS pct
FROM books b JOIN loans l ON l.book_id = b.book_id
WHERE b.condition = 'good' AND l.return_date IS NULL
HAVING COUNT(*) > 0;
```

### Approach 6 — subqueries
```sql
SELECT
  (SELECT COUNT(*) FROM books b JOIN loans l ON l.book_id = b.book_id
   WHERE b.condition = 'good' AND l.return_date IS NULL) AS good_unreturned,
  100.0 * (SELECT COUNT(*) FROM books b JOIN loans l ON l.book_id = b.book_id
           WHERE b.condition = 'good' AND l.return_date IS NULL AND l.renewal_count > 2)
        / NULLIF((SELECT COUNT(*) FROM books b JOIN loans l ON l.book_id = b.book_id
                 WHERE b.condition = 'good' AND l.return_date IS NULL), 0) AS pct_renewed;
```

### Approach 7 — EXISTS
```sql
SELECT
  COUNT(*) AS good_unreturned,
  100.0 * SUM(CASE WHEN l.renewal_count > 2 THEN 1 ELSE 0 END) / COUNT(*)
FROM books b
WHERE b.condition = 'good'
  AND EXISTS (SELECT 1 FROM loans l WHERE l.book_id = b.book_id AND l.return_date IS NULL);
```

### Approach 8 — GROUP BY + conditional counts
```sql
SELECT
  COUNT(*) AS good_unreturned,
  100.0 * COUNT(*) FILTER (WHERE l.renewal_count > 2) / COUNT(*) AS pct_renewed
FROM books b, loans l
WHERE b.book_id = l.book_id AND b.condition = 'good' AND l.return_date IS NULL;
```

### Approach 9 — pandas equivalent
```python
import pandas as pd
def metrics(books, loans):
    df = loans.merge(books, on='book_id')
    df = df[(df['condition'] == 'good') & (df['return_date'].isna())]
    total = len(df)
    return total, 100 * (df['renewal_count'] > 2).sum() / total if total else 0
```

### Approach 10 — materialized counts
```sql
-- pre-aggregate, then query the summary table
SELECT total, pct_renewed FROM book_metrics
WHERE condition = 'good' AND status = 'unreturned';
```

**Best:** Approach 1 — single pass with FILTER, clean semantics.

## Problem 28 — 4 tables (author, transaction, customer, payment): total sales + unique paying customers, by payment type

### Approach 1 — joins + GROUP BY
```sql
SELECT p.type AS payment_type,
       SUM(t.amount) AS total_sales,
       COUNT(DISTINCT c.customer_id) AS unique_customers
FROM transactions t
JOIN customers c   ON c.customer_id = t.customer_id
JOIN payments   p ON p.payment_id = t.payment_id
JOIN authors    a ON a.author_id   = t.author_id
GROUP BY p.type
ORDER BY total_sales DESC;
```

### Approach 2 — CTE step-by-step
```sql
WITH enriched AS (
  SELECT t.amount, p.type AS payment_type, c.customer_id
  FROM transactions t
  JOIN payments   p ON p.payment_id = t.payment_id
  JOIN customers  c ON c.customer_id = t.customer_id
)
SELECT payment_type,
       SUM(amount) AS total_sales,
       COUNT(DISTINCT customer_id) AS unique_customers
FROM enriched GROUP BY payment_type;
```

### Approach 3 — subquery aggregation
```sql
SELECT payment_type,
       (SELECT SUM(amount) FROM transactions t
        JOIN payments p ON p.payment_id = t.payment_id
        WHERE p.type = outer_p.type) AS total_sales,
       (SELECT COUNT(DISTINCT customer_id) FROM transactions t
        JOIN payments p ON p.payment_id = t.payment_id
        WHERE p.type = outer_p.type) AS unique_customers
FROM payments outer_p
GROUP BY payment_type;
```

### Approach 4 — window functions over CTE
```sql
WITH base AS (
  SELECT p.type AS payment_type, t.amount, t.customer_id
  FROM transactions t
  JOIN payments p ON p.payment_id = t.payment_id
)
SELECT DISTINCT payment_type,
       SUM(amount)  OVER (PARTITION BY payment_type) AS total_sales,
       COUNT(DISTINCT customer_id) OVER (PARTITION BY payment_type) AS unique_customers
FROM base;
```

### Approach 5 — HAVING + COUNT DISTINCT
```sql
SELECT p.type, SUM(t.amount) AS total_sales, COUNT(DISTINCT t.customer_id) AS unique_customers
FROM transactions t JOIN payments p ON p.payment_id = t.payment_id
GROUP BY p.type
HAVING SUM(t.amount) > 0;
```

### Approach 6 — FILTER
```sql
SELECT p.type,
       SUM(t.amount) AS total_sales,
       COUNT(DISTINCT t.customer_id) AS unique_customers
FROM transactions t JOIN payments p ON p.payment_id = t.payment_id
GROUP BY p.type;
-- (FILTER alternative: same shape)
```

### Approach 7 — UNION ALL by metric
```sql
SELECT type AS payment_type, 'total_sales' AS metric, SUM(amount)::TEXT AS value
FROM transactions t JOIN payments p ON p.payment_id = t.payment_id
GROUP BY type
UNION ALL
SELECT type, 'unique_customers', COUNT(DISTINCT customer_id)::TEXT
FROM transactions t JOIN payments p ON p.payment_id = t.payment_id
GROUP BY type;
```

### Approach 8 — pandas equivalent
```python
import pandas as pd
def metrics(transactions, payments):
    df = transactions.merge(payments, on='payment_id')
    g = df.groupby('type')
    return g.agg(total_sales=('amount','sum'),
                 unique_customers=('customer_id','nunique'))
```

### Approach 9 — inline view alias
```sql
SELECT payment_type, total_sales, unique_customers FROM (
  SELECT p.type AS payment_type,
         SUM(t.amount) AS total_sales,
         COUNT(DISTINCT t.customer_id) AS unique_customers
  FROM transactions t
  JOIN payments p ON p.payment_id = t.payment_id
  JOIN customers c ON c.customer_id = t.customer_id
  GROUP BY p.type
) v ORDER BY total_sales DESC;
```

### Approach 10 — LATERAL aggregation (creative)
```sql
SELECT v.payment_type, v.total_sales, v.unique_customers
FROM payments p
LEFT JOIN LATERAL (
  SELECT SUM(t.amount) AS total_sales,
         COUNT(DISTINCT t.customer_id) AS unique_customers
  FROM transactions t WHERE t.payment_id = p.payment_id
) v ON true;
```

**Best:** Approach 1 — straightforward GROUP BY.

## Problem 29 — Top 10 customers by books purchased in 2019 (excluding refunded sales)

`sales(sale_id, customer_id, book_id, sale_date, refunded)`, `books(book_id, ...)`, `customers(c_id, ...)`

### Approach 1 — JOIN + GROUP BY + ORDER BY + LIMIT
```sql
SELECT c.customer_id, c.name, COUNT(*) AS books_purchased
FROM sales s
JOIN customers c ON c.customer_id = s.customer_id
WHERE s.sale_date BETWEEN '2019-01-01' AND '2019-12-31'
  AND s.refunded = FALSE
GROUP BY c.customer_id, c.name
ORDER BY books_purchased DESC
LIMIT 10;
```

### Approach 2 — CTE + window
```sql
WITH totals AS (
  SELECT s.customer_id, COUNT(*) AS books_purchased
  FROM sales s
  WHERE s.sale_date BETWEEN '2019-01-01' AND '2019-12-31' AND NOT s.refunded
  GROUP BY s.customer_id
),
ranked AS (
  SELECT customer_id, books_purchased,
         DENSE_RANK() OVER (ORDER BY books_purchased DESC) AS rk
  FROM totals
)
SELECT r.customer_id, c.name, r.books_purchased
FROM ranked r JOIN customers c ON c.customer_id = r.customer_id
WHERE r.rk <= 10;
```

### Approach 3 — HAVING to filter
```sql
SELECT s.customer_id, COUNT(*) c
FROM sales s
WHERE s.sale_date BETWEEN '2019-01-01' AND '2019-12-31' AND NOT s.refunded
GROUP BY s.customer_id
HAVING COUNT(*) >= 1
ORDER BY c DESC LIMIT 10;
```

### Approach 4 — subquery in WHERE
```sql
SELECT * FROM customers WHERE customer_id IN (
  SELECT customer_id FROM sales
  WHERE sale_date BETWEEN '2019-01-01' AND '2019-12-31' AND NOT refunded
  GROUP BY customer_id
  ORDER BY COUNT(*) DESC LIMIT 10
);
```

### Approach 5 — LEFT JOIN with refund filter
```sql
SELECT s.customer_id, COUNT(*) c
FROM sales s LEFT JOIN refunds r ON r.sale_id = s.sale_id
WHERE s.sale_date BETWEEN '2019-01-01' AND '2019-12-31' AND r.sale_id IS NULL
GROUP BY s.customer_id ORDER BY c DESC LIMIT 10;
```

### Approach 6 — NOT IN on refunds
```sql
SELECT customer_id, COUNT(*) c FROM sales
WHERE sale_date BETWEEN '2019-01-01' AND '2019-12-31'
  AND sale_id NOT IN (SELECT sale_id FROM refunds)
GROUP BY customer_id ORDER BY c DESC LIMIT 10;
```

### Approach 7 — window + filter
```sql
SELECT * FROM (
  SELECT s.customer_id, COUNT(*) c,
         ROW_NUMBER() OVER (ORDER BY COUNT(*) DESC) AS rk
  FROM sales s WHERE s.sale_date BETWEEN '2019-01-01' AND '2019-12-31' AND NOT s.refunded
  GROUP BY s.customer_id
) t WHERE rk <= 10;
```

### Approach 8 — pandas equivalent
```python
import pandas as pd
def top10(sales):
    df = sales[(sales['sale_date'].between('2019-01-01','2019-12-31')) & (~sales['refunded'])]
    return (df.groupby('customer_id').size()
              .sort_values(ascending=False).head(10).reset_index(name='c'))
```

### Approach 9 — DISTINCT ON
```sql
SELECT DISTINCT ON (s.customer_id) s.customer_id, COUNT(*) OVER (PARTITION BY s.customer_id) AS c
FROM sales s
WHERE s.sale_date BETWEEN '2019-01-01' AND '2019-12-31' AND NOT s.refunded
ORDER BY c DESC LIMIT 10;
```

### Approach 10 — pre-aggregated materialized view + LIMIT
```sql
SELECT customer_id, books_purchased FROM customer_yearly_summary
WHERE year = 2019 AND NOT refunded
ORDER BY books_purchased DESC LIMIT 10;
```

**Best:** Approach 2 — CTE + DENSE_RANK handles ties transparently.

## Problem 30 — Avg purchase by membership × season (in-season = after Nov 15)

### Approach 1 — CASE in GROUP BY
```sql
SELECT m.membership,
       CASE WHEN EXTRACT(MONTH FROM s.sale_date) >= 11
            THEN 'in_season' ELSE 'out_of_season' END AS season,
       AVG(s.amount) AS avg_purchase
FROM sales s
JOIN customers c ON c.customer_id = s.customer_id
JOIN memberships m ON m.customer_id = c.customer_id
GROUP BY m.membership, season;
```

### Approach 2 — CTE for season
```sql
WITH enriched AS (
  SELECT c.membership, s.amount,
         CASE WHEN s.sale_date >= date_trunc('year', s.sale_date) + INTERVAL '10 months 14 days'
              THEN 'in_season' ELSE 'out_of_season' END AS season
  FROM sales s JOIN customers c ON c.customer_id = s.customer_id
)
SELECT membership, season, AVG(amount) AS avg_purchase FROM enriched GROUP BY membership, season;
```

### Approach 3 — boolean × boolean cross
```sql
SELECT m.membership,
       (s.sale_date >= date_trunc('year', s.sale_date) + INTERVAL '10 months 14 days') AS in_season,
       AVG(s.amount) AS avg
FROM sales s JOIN customers c ON c.customer_id = s.customer_id
JOIN memberships m ON m.customer_id = c.customer_id
GROUP BY m.membership, in_season;
```

### Approach 4 — FILTER
```sql
SELECT m.membership,
       AVG(s.amount) FILTER (WHERE EXTRACT(MONTH FROM s.sale_date) >= 11) AS in_season_avg,
       AVG(s.amount) FILTER (WHERE EXTRACT(MONTH FROM s.sale_date) <  11) AS out_of_season_avg
FROM sales s JOIN customers c ON c.customer_id = s.customer_id
JOIN memberships m ON m.customer_id = c.customer_id
GROUP BY m.membership;
```

### Approach 5 — UNION ALL of two queries
```sql
SELECT m.membership, 'in_season' AS season, AVG(s.amount) avg
FROM sales s JOIN customers c ON c.customer_id = s.customer_id
JOIN memberships m ON m.customer_id = c.customer_id
WHERE EXTRACT(MONTH FROM s.sale_date) >= 11
GROUP BY m.membership
UNION ALL
SELECT m.membership, 'out_of_season', AVG(s.amount) avg
FROM sales s JOIN customers c ON c.customer_id = s.customer_id
JOIN memberships m ON m.customer_id = c.customer_id
WHERE EXTRACT(MONTH FROM s.sale_date) < 11
GROUP BY m.membership;
```

### Approach 6 — subquery + window
```sql
SELECT DISTINCT m.membership, season,
       AVG(amount) OVER (PARTITION BY m.membership, season) AS avg
FROM (
  SELECT m.membership, s.amount,
         CASE WHEN s.sale_date >= date_trunc('year', s.sale_date) + INTERVAL '10 months 14 days'
              THEN 'in' ELSE 'out' END AS season
  FROM sales s JOIN customers c ON c.customer_id = s.customer_id
  JOIN memberships m ON m.customer_id = c.customer_id
) t;
```

### Approach 7 — pandas equivalent
```python
import pandas as pd
def avg_by_mem_season(sales, customers):
    df = sales.merge(customers[['customer_id','membership']], on='customer_id')
    df['season'] = df['sale_date'].dt.month >= 11
    df['season'] = df['season'].map({True: 'in_season', False: 'out_of_season'})
    return df.groupby(['membership','season'])['amount'].mean().reset_index()
```

### Approach 8 — GROUPING SETS
```sql
SELECT m.membership,
       CASE WHEN EXTRACT(MONTH FROM s.sale_date) >= 11 THEN 'in' ELSE 'out' END AS season,
       AVG(s.amount) AS avg
FROM sales s JOIN customers c ON c.customer_id = s.customer_id
JOIN memberships m ON m.customer_id = c.customer_id
GROUP BY GROUPING SETS ((m.membership, season), (m.membership), ());
```

### Approach 9 — generate_series boundary trick
```sql
WITH cutoffs AS (
  SELECT date_trunc('year', generate_series) AS year_start, ... -- (not used; illustrative)
)
SELECT m.membership, AVG(s.amount) avg
FROM sales s JOIN customers c ON c.customer_id = s.customer_id
JOIN memberships m ON m.customer_id = c.customer_id
WHERE s.sale_date >= '2024-11-15' GROUP BY m.membership;
```

### Approach 10 — HAVING + season join
```sql
SELECT m.membership, season, AVG(s.amount) avg
FROM sales s
JOIN customers c ON c.customer_id = s.customer_id
JOIN memberships m ON m.customer_id = c.customer_id
JOIN (VALUES ('in_season', TRUE), ('out_of_season', FALSE)) AS season_map(season, predicate)
  ON predicate = (s.sale_date >= date_trunc('year', s.sale_date) + INTERVAL '10 months 14 days')
GROUP BY m.membership, season;
```

**Best:** Approach 4 — FILTER with two averages on one row is interview-friendly.

## Problem 31 — % revenue from top 5 most expensive books

`books(book_id, title, price)`, `sales(book_id, amount)`

### Approach 1 — CTE + window
```sql
WITH ranked AS (
  SELECT book_id, price, DENSE_RANK() OVER (ORDER BY price DESC) AS rk
  FROM books
)
SELECT 100.0 * SUM(b.price * s.qty) / SUM(s.amount) AS pct_top5_revenue
FROM sales s JOIN books b USING (book_id)
WHERE b.book_id IN (SELECT book_id FROM ranked WHERE rk <= 5);
```

### Approach 2 — subquery + IN
```sql
SELECT 100.0 * SUM(s.amount) / (SELECT SUM(amount) FROM sales) AS pct
FROM sales s WHERE s.book_id IN (
  SELECT book_id FROM books ORDER BY price DESC LIMIT 5
);
```

### Approach 3 — JOIN with ranks
```sql
WITH ranked AS (
  SELECT book_id, ROW_NUMBER() OVER (ORDER BY price DESC) AS rk FROM books
)
SELECT 100.0 * SUM(s.amount) / (SELECT SUM(amount) FROM sales) AS pct
FROM sales s JOIN ranked r USING (book_id) WHERE r.rk <= 5;
```

### Approach 4 — HAVING + subquery
```sql
SELECT 100.0 * SUM(amount) / (SELECT SUM(amount) FROM sales)
FROM sales WHERE book_id IN (
  SELECT book_id FROM books b
  WHERE (SELECT COUNT(DISTINCT price) FROM books b2 WHERE b2.price > b.price) < 5
);
```

### Approach 5 — UNION ALL with aggregation
```sql
WITH top5 AS (SELECT book_id FROM books ORDER BY price DESC LIMIT 5),
totals AS (
  SELECT (SELECT SUM(amount) FROM sales WHERE book_id IN (SELECT book_id FROM top5)) AS top_sum,
         (SELECT SUM(amount) FROM sales) AS total_sum
)
SELECT 100.0 * top_sum / total_sum FROM totals;
```

### Approach 6 — lateral
```sql
SELECT 100.0 * t.top_sum / t.total_sum
FROM (SELECT SUM(amount) FILTER (WHERE book_id IN (
            SELECT book_id FROM books ORDER BY price DESC LIMIT 5)) AS top_sum,
             SUM(amount) AS total_sum FROM sales) t;
```

### Approach 7 — pandas
```python
import pandas as pd
def pct_top5(sales, books):
    top5 = books.nlargest(5, 'price')['book_id']
    total = sales['amount'].sum()
    top_sum = sales[sales['book_id'].isin(top5)]['amount'].sum()
    return 100 * top_sum / total
```

### Approach 8 — window function in WHERE
```sql
SELECT 100.0 * SUM(CASE WHEN rk <= 5 THEN amount END) / SUM(amount) AS pct
FROM (
  SELECT s.amount, DENSE_RANK() OVER (ORDER BY b.price DESC) AS rk
  FROM sales s JOIN books b USING (book_id)
) t;
```

### Approach 9 — RANK + window
```sql
WITH book_sales AS (
  SELECT b.book_id, SUM(s.amount) AS revenue, b.price
  FROM books b LEFT JOIN sales s USING (book_id)
  GROUP BY b.book_id, b.price
),
ranked AS (
  SELECT *, RANK() OVER (ORDER BY price DESC) AS rk FROM book_sales
)
SELECT 100.0 * SUM(revenue) FILTER (WHERE rk <= 5) / SUM(revenue) FROM ranked;
```

### Approach 10 — inline subquery
```sql
SELECT 100.0 *
  (SELECT SUM(amount) FROM sales WHERE book_id IN (SELECT book_id FROM books ORDER BY price DESC LIMIT 5))
  / (SELECT SUM(amount) FROM sales) AS pct;
```

```sql
  FROM sales s JOIN promotions p ON p.promo_id = s.promo_id
```
