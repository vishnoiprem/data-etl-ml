# A working AI agent in 100 lines — taught in Lesson 8.2
from openai import OpenAI

client = OpenAI()

def run_agent(query, max_iter=5):
    messages = [
        {"role": "system", "content": "You are an AI assistant with tools."},
        {"role": "user", "content": query}
    ]

    for i in range(max_iter):
        response = client.chat.completions.create(
            model="gpt-4o-mini",
            messages=messages,
            tools=TOOLS,
            tool_choice="auto"
        )

        if not response.tool_calls:
            return response.choices[0].message.content

        # Execute tools, add to messages, loop again
        for call in response.tool_calls:
            result = exec(call.function.name, **call.args)
            messages.append({"role": "tool", "content": result})

print(run_agent("What's the weather in Tokyo?"))
