# SCB AML Platform — Airflow Dependencies (separate install)
#
# Airflow is installed in a SEPARATE virtualenv because:
#   1. Airflow's pin on Flask/Jinja2/SQLAlchemy versions clashes
#      with the main platform's pinned deps.
#   2. Airflow's scheduler is a long-running daemon — keeping it
#      isolated reduces blast radius if the main env is rebuilt.
#
# Setup (Airflow 3.x):
#   python -m venv venv-airflow
#   source venv-airflow/bin/activate
#   pip install -r requirements-airflow.txt
#   export AIRFLOW_HOME=~/airflow
#   airflow db migrate
#   cp orchestration/airflow_dag.py $AIRFLOW_HOME/dags/
#   airflow api-server --port 8080 &   # 3.x: API server (was webserver)
#   airflow scheduler &
#
# Note: airflow_dag.py uses try/except imports so it still parses
# outside an Airflow env, but it WILL NOT run the pipeline without
# Apache Airflow installed.
